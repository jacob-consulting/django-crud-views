runs = [json.loads((out / (name + ".json")).read_text()) for name in commands]

assert len({r["source_sha256"] for r in runs}) == 1

paths = set(subprocess.check_output(["git", "ls-files", "-z", "--cached", "--others", "--exclude-standard", "--", "src", "tests", "docs", "noxfile.py", "pyproject.toml", "mkdocs.yml"]).decode().split("\0")) - {""}

assert all(hashlib.sha256((root / p).read_bytes()).hexdigest() == h for p, h in manifest.items())

test_evidence = {}
for name in ("full-normal", "full-recorder", "supported-4-2", "supported-6-0"):
    cases = list(ET.parse(out / (name + ".xml")).iter("testcase"))
    matched = {prefix: [c.attrib for c in cases if c.get("name", "").startswith(prefix)] for prefix in critical}
    assert all(matched.values()), (name, [p for p, rows in matched.items() if not rows])
    assert all(not list(c) for c in cases if any(c.get("name", "").startswith(p) for p in critical))
    test_evidence[name] = {"collected":len(cases), "critical_cases":matched, "skips":[{**c.attrib,"reason":c.find("skipped").get("message")} for c in cases if c.find("skipped") is not None]}
browser_path = phase / "06.34-10-browser-evidence.json"
browser = json.loads(browser_path.read_text())
assert browser["status"] == "passed" and len(browser["cases"]) == 63
assert all(all(c["facts"]["presentation"].values()) for c in browser["cases"])
ui_checks = {}
for case in browser["cases"]:
    for assertion in case["assertions"]:
        if any(word in assertion["name"] for word in ("collapsed", "actor-neutral", "error belongs")):
            assert assertion["actual"] == assertion["expected"]
            ui_checks.setdefault(assertion["name"], []).append(case["id"])
